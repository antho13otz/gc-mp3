cmake_minimum_required(VERSION 3.18)
project(gbahote C)
set(MGBA_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../android/app/src/main/cpp/mgba)
set(M_CORE_GBA ON)
set(M_CORE_GB OFF)
set(USE_DEBUGGERS OFF)
set(USE_GDB_STUB OFF)
set(USE_ELF OFF)
set(USE_ZLIB OFF)
set(USE_PNG OFF)
set(USE_LZMA OFF)
set(USE_SQLITE3 OFF)
set(USE_LIBZIP OFF)
set(USE_MINIZIP OFF)
set(USE_FFMPEG OFF)
set(USE_LUA OFF)
set(ENABLE_SCRIPTING OFF)
set(USE_DISCORD_RPC OFF)
set(BUILD_GL OFF)
set(BUILD_GLES2 OFF)
set(BUILD_GLES3 OFF)
set(USE_EPOXY OFF)
set(USE_EDITLINE OFF)
set(BUILD_QT OFF)
set(BUILD_SDL OFF)
set(SKIP_GIT ON)
add_subdirectory(${MGBA_DIR} ${CMAKE_CURRENT_BINARY_DIR}/mgba EXCLUDE_FROM_ALL)
add_executable(hote hote_joybus.c)
target_include_directories(hote PRIVATE ${MGBA_DIR}/include ${MGBA_DIR}/src ${CMAKE_CURRENT_BINARY_DIR}/mgba/include)
target_link_libraries(hote mgba)
